"use client";

import {useEffect, useState} from "react";

type Stock = {id: string; status: "loading" | "ready" | "unknown" | "error"; quantity?: number};
const cache = new Map<string, {value: Stock; expires: number}>();
const pending = new Map<string, ((stock: Stock) => void)[]>();
let timer: ReturnType<typeof setTimeout> | undefined;

function readStock(id: string): Promise<Stock> {
  const saved = cache.get(id);
  if (saved && saved.expires > Date.now()) return Promise.resolve(saved.value);
  return new Promise(resolve => {
    const waiting = pending.get(id);
    if (waiting) waiting.push(resolve); else pending.set(id, [resolve]);
    if (timer) return;
    timer = setTimeout(async () => {
      timer = undefined;
      const work = [...pending.entries()];
      pending.clear();
      for (let start = 0; start < work.length; start += 100) {
        const batch = work.slice(start, start + 100);
        let data: {product_id: string; quantity: number}[] | null = null;
        try {
          const response = await fetch(`/api/crm-stock?ids=${batch.map(([key]) => key).join(",")}`, {cache: "no-store", signal: AbortSignal.timeout(6000)});
          if (response.ok) data = (await response.json()).data;
          if (!Array.isArray(data)) data = null;
        } catch { /* An unavailable API must never become zero stock. */ }
        for (const [key, listeners] of batch) {
          const row = data?.find(item => item.product_id === key);
          const value: Stock = row ? {id: key, status: "ready", quantity: row.quantity} : {id: key, status: data ? "unknown" : "error"};
          if (cache.size > 2000) cache.clear();
          cache.set(key, {value, expires: Date.now() + 15000});
          listeners.forEach(listener => listener(value));
        }
      }
    }, 20);
  });
}

export function useCrmStock(id: string) {
  const [stock, setStock] = useState<Stock>({id, status: "loading"});
  useEffect(() => {
    let alive = true;
    if (!/^[1-9]\d{0,14}$/.test(id)) return;
    const refresh = () => {
      if (document.visibilityState === "hidden") return;
      void readStock(id).then(value => { if (alive) setStock(value); });
    };
    refresh();
    const interval = setInterval(refresh, 30000);
    document.addEventListener("visibilitychange", refresh);
    return () => { alive = false; clearInterval(interval); document.removeEventListener("visibilitychange", refresh); };
  }, [id]);
  return stock.id === id ? stock : {id, status: "loading" as const};
}

export function StockMessage({stock, compact = false}: {stock: Stock; compact?: boolean}) {
  if (stock.status === "loading") return compact ? null : <p className="text-sm text-neutral-500" role="status">Үлдэгдэл шалгаж байна…</p>;
  if (stock.status !== "ready") return compact ? null : <p className="rounded-xl border border-neutral-200 p-3 text-sm text-neutral-700">Үлдэгдлийг <a className="font-bold underline" href="tel:72005588">72005588</a> дугаараар лавлана уу.</p>;
  if (stock.quantity === 0) return <p className="rounded-xl border border-amber-200 bg-amber-50 p-3 text-sm leading-relaxed text-amber-950">Үлдэгдэл цөөн байгаа тул та <a className="font-bold underline" href="tel:72005588">72005588</a> дугаараар холбогдож асууна уу.</p>;
  return <p className="text-sm font-medium text-emerald-800">Бэлэн: {stock.quantity} ширхэг</p>;
}

export function CrmStock({id}: {id: string}) {
  return <StockMessage stock={useCrmStock(id)} compact />;
}
