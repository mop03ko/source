"use client";

import {useEffect, useState} from "react";
import {MapPin, PackageCheck, Phone} from "lucide-react";

type Branch = {name: string; quantity: number};
type Stock = {id: string; status: "loading" | "ready" | "unknown" | "error"; quantity?: number; branches?: Branch[]};
const cache = new Map<string, {value: Stock; expires: number}>();
const pending = new Map<string, ((stock: Stock) => void)[]>();
let timer: ReturnType<typeof setTimeout> | undefined;

function readStock(id: string): Promise<Stock> {
  const saved = cache.get(id);
  if (saved && saved.expires > Date.now()) return Promise.resolve(saved.value);
  return new Promise(resolve => {
    const waiting = pending.get(id);
    if (waiting) waiting.push(resolve); else pending.set(id, [resolve]);
    if (timer) return;
    timer = setTimeout(async () => {
      timer = undefined;
      const work = [...pending.entries()];
      pending.clear();
      for (let start = 0; start < work.length; start += 100) {
        const batch = work.slice(start, start + 100);
        let data: {product_id: string; quantity: number; branches?: Branch[]}[] | null = null;
        try {
          const response = await fetch(`/api/crm-stock?ids=${batch.map(([key]) => key).join(",")}`, {cache: "no-store", signal: AbortSignal.timeout(6000)});
          if (response.ok) data = (await response.json()).data;
          if (!Array.isArray(data)) data = null;
        } catch { /* An unavailable API must never become zero stock. */ }
        for (const [key, listeners] of batch) {
          const row = data?.find(item => item.product_id === key);
          const value: Stock = row ? {id: key, status: "ready", quantity: row.quantity, branches: row.branches} : {id: key, status: data ? "unknown" : "error"};
          if (cache.size > 2000) cache.clear();
          cache.set(key, {value, expires: Date.now() + 15000});
          listeners.forEach(listener => listener(value));
        }
      }
    }, 20);
  });
}

export function useCrmStock(id: string) {
  const [stock, setStock] = useState<Stock>({id, status: "loading"});
  useEffect(() => {
    let alive = true;
    if (!/^[1-9]\d{0,14}$/.test(id)) return;
    const refresh = () => {
      if (document.visibilityState === "hidden") return;
      void readStock(id).then(value => { if (alive) setStock(value); });
    };
    refresh();
    const interval = setInterval(refresh, 30000);
    document.addEventListener("visibilitychange", refresh);
    return () => { alive = false; clearInterval(interval); document.removeEventListener("visibilitychange", refresh); };
  }, [id]);
  return stock.id === id ? stock : {id, status: "loading" as const};
}

export function StockMessage({stock, compact = false}: {stock: Stock; compact?: boolean}) {
  if (stock.status === "loading") return compact ? null : <p className="text-sm text-neutral-500" role="status">Үлдэгдэл шалгаж байна…</p>;
  if (stock.status !== "ready") return compact ? null : <p className="rounded-xl border border-neutral-200 p-3 text-sm text-neutral-700">Үлдэгдлийг <a className="font-bold underline" href="tel:72005588">72005588</a> дугаараар лавлана уу.</p>;
  const available = stock.branches?.filter(branch => branch.quantity > 0) ?? [];
  if (compact && (stock.quantity ?? 0) > 0) return <div className="rounded-lg bg-emerald-50 px-2.5 py-2 text-xs text-emerald-900"><p className="font-semibold">Бэлэн · {stock.quantity} ширхэг</p>{available.length > 0 && <p className="mt-1 leading-relaxed">{available.map(branch => `${branch.name} (${branch.quantity})`).join(" · ")}</p>}</div>;
  if (stock.quantity === 0) return <div className="rounded-xl border border-amber-200 bg-amber-50 p-4 text-sm leading-relaxed text-amber-950"><p>Үлдэгдэл цөөн байгаа тул та <a className="font-bold underline" href="tel:72005588">72005588</a> дугаараар холбогдож асууна уу.</p>{!compact && <a href="tel:72005588" className="mt-3 inline-flex min-h-11 items-center gap-2 rounded-lg border border-amber-300 bg-white px-4 font-semibold"><Phone className="h-4 w-4" aria-hidden="true" />Залгаж лавлах</a>}</div>;
  return <section aria-label="Салбарын үлдэгдэл" className="overflow-hidden rounded-2xl border border-emerald-200 bg-white">
    <div className="flex flex-wrap items-center justify-between gap-3 bg-emerald-50 px-4 py-3.5">
      <div className="flex items-center gap-2.5"><PackageCheck className="h-5 w-5 text-emerald-700" aria-hidden="true" /><h2 className="text-sm font-semibold text-emerald-950">Бэлэн байгаа салбар</h2></div>
      <span className="rounded-full bg-white px-3 py-1 text-sm font-bold text-emerald-800">Нийт {stock.quantity} ширхэг</span>
    </div>
    <ul className="divide-y divide-neutral-100 px-4">{(stock.branches ?? []).map(branch => <li key={branch.name} className="flex items-center justify-between gap-3 py-3">
      <span className="flex min-w-0 items-center gap-2 text-sm text-neutral-800"><MapPin className="h-4 w-4 shrink-0 text-neutral-400" aria-hidden="true" />{branch.name}</span>
      <span className={`shrink-0 rounded-md px-2.5 py-1 text-xs font-semibold ${branch.quantity > 0 ? "bg-emerald-50 text-emerald-800" : "bg-neutral-100 text-neutral-500"}`}>{branch.quantity > 0 ? `${branch.quantity} ширхэг` : "Одоогоор байхгүй"}</span>
    </li>)}</ul>
    <div className="flex flex-wrap items-center justify-between gap-2 border-t border-neutral-100 px-4 py-3 text-xs text-neutral-500"><span>Очихоосоо өмнө үлдэгдлийг лавлаарай.</span><a href="tel:72005588" className="inline-flex min-h-9 items-center gap-1.5 font-semibold text-neutral-800 underline underline-offset-4"><Phone className="h-3.5 w-3.5" aria-hidden="true" />72005588</a></div>
  </section>;
}

export function CrmStock({id}: {id: string}) {
  return <StockMessage stock={useCrmStock(id)} compact />;
}
